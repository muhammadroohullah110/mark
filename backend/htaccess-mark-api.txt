# ============================================================
# PASTE THIS INTO: public_html/mark-api/.htaccess
# This reverse-proxies requests to the FastAPI backend
# ============================================================

RewriteEngine On
RewriteCond %{REQUEST_FILENAME} !-f
RewriteCond %{REQUEST_FILENAME} !-d

# Proxy all requests to uvicorn running on port 8090
RewriteRule ^(.*)$ http://127.0.0.1:8090/$1 [P,L]
